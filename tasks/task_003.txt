# Implementation for agent_registry_service/core/state_machine.py

from enum import Enum
from typing import Dict, List, Optional, Set, Tuple
import logging
from datetime import datetime
import threading

logger = logging.getLogger(__name__)

class AgentState(Enum):
    NEW = "new"
    EXPERIMENTAL = "experimental"
    CANDIDATE = "candidate"
    STABLE = "stable"
    DEPRECATED = "deprecated"
    ARCHIVED = "archived"
    KILLED = "killed"

# Define transition matrix with required fields and validation rules
ALLOWED_TRANSITIONS = {
    AgentState.NEW: {
        AgentState.EXPERIMENTAL: {
            "required_fields": ["version", "owner", "test_results"],
            "validators": ["has_passing_tests", "has_documentation"]
        }
    },
    AgentState.EXPERIMENTAL: {
        AgentState.CANDIDATE: {
            "required_fields": ["performance_metrics", "stability_report"],
            "validators": ["meets_performance_threshold", "has_sufficient_test_coverage"]
        },
        AgentState.KILLED: {
            "required_fields": ["reason", "approved_by"],
            "validators": ["has_approval_from_owner"]
        }
    },
    AgentState.CANDIDATE: {
        AgentState.STABLE: {
            "required_fields": ["approval_votes", "security_review", "compliance_check"],
            "validators": ["has_minimum_approval_votes", "passes_security_review"]
        },
        AgentState.EXPERIMENTAL: {
            "required_fields": ["reason", "issues_found"],
            "validators": []
        },
        AgentState.KILLED: {
            "required_fields": ["reason", "approved_by"],
            "validators": ["has_approval_from_owner"]
        }
    },
    AgentState.STABLE: {
        AgentState.DEPRECATED: {
            "required_fields": ["reason", "replacement_agent", "timeline"],
            "validators": ["has_valid_replacement"]
        }
    },
    AgentState.DEPRECATED: {
        AgentState.ARCHIVED: {
            "required_fields": ["usage_statistics", "migration_report"],
            "validators": ["usage_below_threshold"]
        }
    },
    AgentState.ARCHIVED: {
        AgentState.KILLED: {
            "required_fields": ["reason", "data_retention_plan"],
            "validators": ["has_data_migration_plan"]
        }
    }
    # Note: KILLED is a terminal state with no outgoing transitions
}

class TransitionValidationError(Exception):
    """Raised when transition validation fails"""
    def __init__(self, from_state, to_state, reason, missing_fields=None, validation_errors=None):
        self.from_state = from_state
        self.to_state = to_state
        self.reason = reason
        self.missing_fields = missing_fields or []
        self.validation_errors = validation_errors or []
        
        message = f"Cannot transition from {from_state.value} to {to_state.value}: {reason}"
        super().__init__(message)

class StateMachine:
    def __init__(self, initial_state: AgentState = AgentState.NEW):
        self._current_state = initial_state
        self._state_history = [(initial_state, {}, datetime.now())]
        self._lock = threading.RLock()  # Reentrant lock for thread safety
        self._validators = {
            "has_passing_tests": self._validate_passing_tests,
            "has_documentation": self._validate_documentation,
            "meets_performance_threshold": self._validate_performance,
            "has_sufficient_test_coverage": self._validate_test_coverage,
            "has_approval_from_owner": self._validate_owner_approval,
            "has_minimum_approval_votes": self._validate_approval_votes,
            "passes_security_review": self._validate_security_review,
            "has_valid_replacement": self._validate_replacement,
            "usage_below_threshold": self._validate_usage_threshold,
            "has_data_migration_plan": self._validate_data_migration
        }
    
    @property
    def current_state(self) -> AgentState:
        """Get the current state of the agent"""
        return self._current_state
    
    @property
    def state_history(self) -> List[Tuple[AgentState, Dict, datetime]]:
        """Get the history of state transitions"""
        return self._state_history.copy()
    
    def can_transition_to(self, to_state: AgentState) -> bool:
        """Check if transition to the given state is allowed from current state"""
        return to_state in ALLOWED_TRANSITIONS.get(self._current_state, {})
    
    def validate_transition(self, to_state: AgentState, context: Dict) -> Tuple[bool, Optional[TransitionValidationError]]:
        """Validate if transition is allowed and context contains required fields"""
        # Check if transition is allowed
        if not self.can_transition_to(to_state):
            return False, TransitionValidationError(
                self._current_state, 
                to_state, 
                "Transition not allowed"
            )
        
        transition_rules = ALLOWED_TRANSITIONS[self._current_state][to_state]
        
        # Check required fields
        required_fields = transition_rules["required_fields"]
        missing_fields = [field for field in required_fields if field not in context]
        if missing_fields:
            return False, TransitionValidationError(
                self._current_state,
                to_state,
                "Missing required fields",
                missing_fields=missing_fields
            )
        
        # Run validators
        validation_errors = []
        for validator_name in transition_rules["validators"]:
            validator_func = self._validators.get(validator_name)
            if validator_func and not validator_func(context):
                validation_errors.append(validator_name)
        
        if validation_errors:
            return False, TransitionValidationError(
                self._current_state,
                to_state,
                "Failed validation checks",
                validation_errors=validation_errors
            )
        
        return True, None
    
    def transition(self, to_state: AgentState, context: Dict) -> bool:
        """
        Attempt to transition to the given state with the provided context
        Returns True if transition was successful, False otherwise
        """
        with self._lock:
            is_valid, error = self.validate_transition(to_state, context)
            if not is_valid:
                logger.warning(f"Invalid transition attempted: {error}")
                return False
            
            # Perform the transition
            self._current_state = to_state
            self._state_history.append((to_state, context.copy(), datetime.now()))
            logger.info(f"Transitioned to {to_state.value} state")
            return True
    
    # Validator implementations
    def _validate_passing_tests(self, context: Dict) -> bool:
        test_results = context.get("test_results", {})
        return test_results.get("passing", 0) > 0 and test_results.get("failing", 0) == 0
    
    def _validate_documentation(self, context: Dict) -> bool:
        return bool(context.get("documentation_url"))
    
    def _validate_performance(self, context: Dict) -> bool:
        metrics = context.get("performance_metrics", {})
        return metrics.get("latency_ms", float('inf')) < 500 and metrics.get("success_rate", 0) > 0.95
    
    def _validate_test_coverage(self, context: Dict) -> bool:
        return context.get("stability_report", {}).get("test_coverage", 0) > 80
    
    def _validate_owner_approval(self, context: Dict) -> bool:
        return context.get("approved_by") == context.get("owner")
    
    def _validate_approval_votes(self, context: Dict) -> bool:
        return len(context.get("approval_votes", [])) >= 3
    
    def _validate_security_review(self, context: Dict) -> bool:
        return context.get("security_review", {}).get("passed", False)
    
    def _validate_replacement(self, context: Dict) -> bool:
        return bool(context.get("replacement_agent"))
    
    def _validate_usage_threshold(self, context: Dict) -> bool:
        return context.get("usage_statistics", {}).get("active_users", float('inf')) < 10
    
    def _validate_data_migration(self, context: Dict) -> bool:
        return bool(context.get("data_retention_plan"))

```python
@router.put("/{agent_id}/state", response_model=schemas.Agent)
async def update_agent_state(
    agent_id: UUID,
    state_update: schemas.AgentStateUpdate,
    db: AsyncSession = Depends(deps.get_db),
    current_user: models.User = Depends(deps.get_current_active_user),
) -> Any:
    """
    Update an agent's state.
    
    Requires:
    - Valid agent_id
    - Valid target state (must be a legal transition from current state)
    - Justification for state change
    - Appropriate permissions
    """
    # Check if agent exists
    agent = await crud.agent.get(db=db, id=agent_id)
    if not agent:
        raise HTTPException(status_code=404, detail="Agent not found")
    
    # Check permissions
    if not crud.agent.can_update_state(db, current_user, agent):
        raise HTTPException(
            status_code=403, 
            detail="Not enough permissions to change this agent's state"
        )
    
    # Validate state transition
    try:
        updated_agent = await crud.agent.update_state(
            db=db,
            db_obj=agent,
            target_state=state_update.target_state,
            justification=state_update.justification,
            updated_by=current_user.id
        )
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))
    
    # Log the state transition
    await crud.agent_state_log.create(
        db=db,
        obj_in=schemas.AgentStateLogCreate(
            agent_id=agent_id,
            previous_state=agent.state,
            new_state=updated_agent.state,
            justification=state_update.justification,
            changed_by=current_user.id
        )
    )
    
    return updated_agent
```

from datetime import datetime

from agent_registry_service.core.state_machine import StateMachine, InvalidTransitionError

class TestStateMachine:
    def test_initial_state(self):
        sm = StateMachine(initial_state="PENDING", allowed_transitions={
            "PENDING": ["ACTIVE"],
            "ACTIVE": ["INACTIVE", "TERMINATED"],
            "INACTIVE": ["ACTIVE", "TERMINATED"],
            "TERMINATED": []
        })

        assert sm.state_history[0]["state"] == "PENDING"
        assert isinstance(sm.state_history[0]["timestamp"], datetime)
        
    def test_valid_transition(self):
        sm = StateMachine(initial_state="PENDING", allowed_transitions={
            "PENDING": ["ACTIVE"],
            "ACTIVE": ["INACTIVE"]
        })
        
        sm.transition("ACTIVE", metadata={"user_id": "test123"})
        assert sm.current_state == "ACTIVE"
        assert len(sm.state_history) == 2
        assert sm.state_history[1]["state"] == "ACTIVE"
        assert sm.state_history[1]["metadata"]["user_id"] == "test123"
        
    def test_invalid_transition(self):
        sm = StateMachine(initial_state="PENDING", allowed_transitions={
            "PENDING": ["ACTIVE"],
            "ACTIVE": ["INACTIVE"]
        })
        
        with pytest.raises(InvalidTransitionError) as excinfo:
            sm.transition("INACTIVE")
        assert "Cannot transition from PENDING to INACTIVE" in str(excinfo.value)
        
    def test_transition_hooks(self):
        hook_called = {"pre": False, "post": False}
        
        def pre_hook(from_state, to_state, metadata):
            hook_called["pre"] = True
            assert from_state == "PENDING"
            assert to_state == "ACTIVE"
            assert metadata.get("test") == "data"
            
        def post_hook(from_state, to_state, metadata):
            hook_called["post"] = True
            assert from_state == "PENDING"
            assert to_state == "ACTIVE"
            assert metadata.get("test") == "data"
            
        sm = StateMachine(initial_state="PENDING", allowed_transitions={
            "PENDING": ["ACTIVE"]
        })

## Overview

The Agent Registry Service implements a state machine to track and manage agent lifecycle states. This document describes the state machine implementation, valid state transitions, and how to interact with agent states through the API.

## State Diagram

```
PENDING ──────► ACTIVE ◄────┐
                  │          │
                  ▼          │
                INACTIVE ────┘
                  │
                  ▼
              TERMINATED

## States

- **PENDING**: Initial state for newly created agents
- **ACTIVE**: Agent is operational and available for use
- **INACTIVE**: Agent is temporarily disabled but can be reactivated
- **TERMINATED**: Agent has been permanently decommissioned

## API Usage

### Update Agent State

```
PUT /agents/{agent_id}/state
```

Request body:
```json

  "state": "ACTIVE",
  "metadata": {

  }
