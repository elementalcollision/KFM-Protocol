// src/auth/authProvider.js
export const authProvider = {
  login: ({ username, password }) => {
    return fetch('https://api.kfm.example.com/auth/token', {
      method: 'POST',
      body: JSON.stringify({ username, password }),
      headers: { 'Content-Type': 'application/json' },
    })
      .then(response => {
        if (response.status < 200 || response.status >= 300) {
          throw new Error(response.statusText);
        }
        return response.json();
      })
      .then(auth => {
        localStorage.setItem('auth', JSON.stringify(auth));
        localStorage.setItem('token', auth.token);
      });
  },
  checkAuth: () => {
    return localStorage.getItem('token') ? Promise.resolve() : Promise.reject();
  },
  logout: () => {
    localStorage.removeItem('token');
    localStorage.removeItem('auth');
    return Promise.resolve();
  },
  getPermissions: () => {
    const auth = JSON.parse(localStorage.getItem('auth') || '{}');
    return Promise.resolve(auth.permissions);
  },
};

// Implementation approach for Dashboard component
import React, { useState, useEffect } from 'react';
import { useDataProvider } from 'react-admin';
import { Card, CardContent, Grid, Typography, Box } from '@mui/material';
import { ResponsiveContainer, PieChart, Pie, Cell, BarChart, Bar, XAxis, YAxis, Tooltip } from 'recharts';
import ReactFlow from 'react-flow-renderer';

// Custom hook for dashboard data
const useDashboardData = () => {
  const dataProvider = useDataProvider();
  const [loading, setLoading] = useState(true);
  const [stats, setStats] = useState(null);
  
  useEffect(() => {
    const fetchData = async () => {
      try {
        const [systemStats, agentStats, recentOps] = await Promise.all([
          dataProvider.getOne('system/health', {}),
          dataProvider.getList('agents/stats', { pagination: { page: 1, perPage: 100 } }),
          dataProvider.getList('operations', { 
            pagination: { page: 1, perPage: 5 },
            sort: { field: 'timestamp', order: 'DESC' } 
          }),
        ]);
        
        setStats({ systemStats, agentStats, recentOps });
      } catch (error) {
        console.error('Error fetching dashboard data:', error);
      } finally {
        setLoading(false);
      }
    };
    
    fetchData();
    // Set up polling interval for real-time updates
    const interval = setInterval(fetchData, 30000); // 30-second refresh
    return () => clearInterval(interval);
  }, [dataProvider]);
  
  return { loading, stats };
};

// Agent dependency visualization component
const AgentDependencyGraph = ({ agents }) => {
  // Transform agent data into nodes and edges for ReactFlow
  const elements = React.useMemo(() => {
    const nodes = agents.map(agent => ({
      id: agent.id,
      data: { label: agent.name },
      position: { x: Math.random() * 500, y: Math.random() * 500 },
      style: { 
        background: agent.state === 'STABLE' ? '#4caf50' : 
                   agent.state === 'EXPERIMENTAL' ? '#ff9800' : '#2196f3',
        border: '1px solid #222138',
        borderRadius: 5,
        padding: 10,
      }
    }));
    
    const edges = [];
    agents.forEach(agent => {
      if (agent.dependencies) {
        agent.dependencies.forEach(depId => {
          edges.push({
            id: `${agent.id}-${depId}`,
            source: agent.id,
            target: depId,
            type: 'smoothstep',
            animated: true,
          });
        });
      }
    });
    
    return [...nodes, ...edges];
  }, [agents]);
  
  return (
    <Box height="400px">
      <ReactFlow 
        elements={elements}
        defaultZoom={1.5}
        minZoom={0.2}
        maxZoom={4}
      />
    </Box>
  );
};

// Permission context setup
const PermissionsContext = React.createContext();

export const PermissionsProvider = ({ children }) => {
  const [permissions, setPermissions] = useState([]);
  const { identity } = useGetIdentity();
  
  useEffect(() => {
    if (identity?.id) {
      authProvider.getPermissions().then(setPermissions);
    }
  }, [identity]);
  
  return (
    <PermissionsContext.Provider value={permissions}>
      {children}
    </PermissionsContext.Provider>
  );
};

// Custom hook for permission checks
export const useCanAccess = (resource, action) => {
  const permissions = useContext(PermissionsContext);
  return permissions.some(p => p.resource === resource && p.action === action);
};
