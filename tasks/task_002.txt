```python
from sqlalchemy.orm import Session
from fastapi.encoders import jsonable_encoder
from app.models.agent import Agent
from app.schemas.agent import AgentCreate, AgentUpdate
import logging
from uuid import UUID
from typing import List, Optional

logger = logging.getLogger(__name__)

def get(db: Session, agent_id: UUID) -> Optional[Agent]:
    logger.info(f"Retrieving agent with ID: {agent_id}")
    return db.query(Agent).filter(Agent.id == agent_id).first()

def get_multi(db: Session, *, skip: int = 0, limit: int = 100) -> List[Agent]:
    logger.info(f"Retrieving multiple agents with skip={skip}, limit={limit}")
    return db.query(Agent).offset(skip).limit(limit).all()

def create(db: Session, *, obj_in: AgentCreate) -> Agent:
    logger.info("Creating new agent")
    obj_in_data = jsonable_encoder(obj_in)
    db_obj = Agent(**obj_in_data)
    db.add(db_obj)
    db.commit()
    db.refresh(db_obj)
    logger.info(f"Created agent with ID: {db_obj.id}")
    return db_obj

def update(db: Session, *, db_obj: Agent, obj_in: AgentUpdate) -> Agent:
    logger.info(f"Updating agent with ID: {db_obj.id}")
    obj_data = jsonable_encoder(db_obj)
    update_data = obj_in.dict(exclude_unset=True)
    
    for field in obj_data:
        if field in update_data:
            setattr(db_obj, field, update_data[field])
    
    db.add(db_obj)
    db.commit()
    db.refresh(db_obj)
    logger.info(f"Updated agent with ID: {db_obj.id}")
    return db_obj

def remove(db: Session, *, agent_id: UUID) -> Agent:
    logger.info(f"Removing agent with ID: {agent_id}")
    obj = db.query(Agent).get(agent_id)
    db.delete(obj)
    db.commit()
    logger.info(f"Removed agent with ID: {agent_id}")
    return obj
```

```python
# middleware/rate_limit.py
from fastapi import Request, HTTPException
import time
from collections import defaultdict

# Simple in-memory rate limiter
class RateLimiter:
    def __init__(self, requests_per_minute: int = 60):
        self.requests_per_minute = requests_per_minute
        self.requests = defaultdict(list)
        
    async def __call__(self, request: Request):
        client_ip = request.client.host
        now = time.time()
        
        # Remove requests older than 1 minute
        self.requests[client_ip] = [req_time for req_time in self.requests[client_ip] 
                                   if now - req_time < 60]
        
        # Check if rate limit exceeded
        if len(self.requests[client_ip]) >= self.requests_per_minute:
            raise HTTPException(status_code=429, detail="Rate limit exceeded")
        
        # Add current request timestamp
        self.requests[client_ip].append(now)
```

### State Transition Validation Logic
```python
# utils/state_validation.py
def validate_state_transition(current_state: str, new_state: str) -> bool:
    """
    Validates if a state transition is allowed based on defined rules.
    
    Valid transitions:
    - created → initializing
    - initializing → ready, error
    - ready → busy, terminated
    - busy → ready, error
    - error → initializing, terminated
    - terminated → (no valid transitions)
    """
    valid_transitions = {
        "created": ["initializing"],
        "initializing": ["ready", "error"],
        "ready": ["busy", "terminated"],
        "busy": ["ready", "error"],
        "error": ["initializing", "terminated"],
        "terminated": []
    }
    
    return new_state in valid_transitions.get(current_state, [])
```

## 3. Request ID Middleware
```python
# app/middleware/request_id.py
import uuid
from contextvars import ContextVar
from starlette.middleware.base import BaseHTTPMiddleware

# Create a context variable to store the request ID
request_id_var = ContextVar("request_id", default=None)

def get_request_id():
    return request_id_var.get()

class RequestIdMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request, call_next):
        # Check if request has X-Request-ID header, otherwise generate one
        request_id = request.headers.get("X-Request-ID", str(uuid.uuid4()))
        
        # Set the request ID in the context
        token = request_id_var.set(request_id)
        
        try:
            # Process the request
            response = await call_next(request)
            # Add the request ID to the response headers
            response.headers["X-Request-ID"] = request_id
            return response
        finally:
            # Reset the context variable
            request_id_var.reset(token)
```

```python
def log_timing(func):
    """Decorator to log function execution time"""
    @wraps(func)
    async def wrapper(*args, **kwargs):
        start_time = time.time()
        result = await func(*args, **kwargs)
        execution_time = time.time() - start_time
        logger.info(f"Performance: {func.__name__} executed in {execution_time:.4f}s")
        return result
    return wrapper
```

```python
from starlette.middleware.base import BaseHTTPMiddleware
import logging

logger = logging.getLogger(__name__)

class RequestTimingMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request, call_next):
        start_time = time.time()
        response = await call_next(request)
        process_time = time.time() - start_time
        
        # Add timing header
        response.headers["X-Process-Time"] = str(process_time)
        
        # Log request timing
        logger.info(
            f"Request timing",
            extra={
                "path": request.url.path,
                "method": request.method,
                "process_time": process_time,
                "status_code": response.status_code
            }
        )
        return response
```

```python
def setup_memory_monitoring(app: FastAPI):

                "vms_mb": memory_info.vms / 1024 / 1024
            }
        )
    
    # Log memory usage every 5 minutes
    scheduler.add_job(log_memory_usage, 'interval', minutes=5)
```
