```python
# agent_registry_service/core/registry_manager.py
import asyncio
from typing import Dict, List, Optional
from uuid import UUID
import logging

from sqlalchemy.ext.asyncio import AsyncSession
from fastapi import Depends

from ..schemas.agent_registry_entry import AgentRegistryEntry
from ..crud.crud_agent_registry import (
    create_agent_entry, get_agent_entry, get_all_agent_entries,
    update_agent_entry, delete_agent_entry
)
from ..db.session import get_db

logger = logging.getLogger(__name__)

class RegistryManager:
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super(RegistryManager, cls).__new__(cls)
            cls._instance._initialized = False
        return cls._instance
    
    def __init__(self):
        if self._initialized:
            return
            
        self._registry: Dict[UUID, AgentRegistryEntry] = {}
        self._lock = asyncio.Lock()
        self._initialized = True
        logger.info("Registry Manager initialized")
    
    async def initialize_from_db(self, db: AsyncSession):
        """Load all agent entries from database into memory on startup"""
        async with self._lock:
            logger.info("Loading agent registry from database")
            entries = await get_all_agent_entries(db)
            self._registry = {entry.id: entry for entry in entries}
            logger.info(f"Loaded {len(self._registry)} agent entries into registry")
    
    async def register_agent(self, agent: AgentRegistryEntry, db: AsyncSession) -> AgentRegistryEntry:
        """Register a new agent in both memory and database"""
        async with self._lock:
            # First persist to database
            db_agent = await create_agent_entry(db, agent)
            # Then update in-memory store
            self._registry[db_agent.id] = db_agent
            return db_agent
    
    async def get_agent(self, agent_id: UUID) -> Optional[AgentRegistryEntry]:
        """Get agent from in-memory store"""
        return self._registry.get(agent_id)
    
    async def get_all_agents(self) -> List[AgentRegistryEntry]:
        """Get all agents from in-memory store"""
        return list(self._registry.values())
    
    async def update_agent(self, agent_id: UUID, agent_data: dict, db: AsyncSession) -> Optional[AgentRegistryEntry]:
        """Update agent in both memory and database"""
        async with self._lock:
            if agent_id not in self._registry:
                return None
                
            # Update in database first
            updated_agent = await update_agent_entry(db, agent_id, agent_data)
            if updated_agent:
                # Then update in-memory store
                self._registry[agent_id] = updated_agent
            return updated_agent
    
    async def deregister_agent(self, agent_id: UUID, db: AsyncSession) -> bool:
        """Remove agent from both memory and database"""
        async with self._lock:
            if agent_id not in self._registry:
                return False
                
            # Delete from database first
            success = await delete_agent_entry(db, agent_id)
            if success:
                # Then remove from in-memory store
                del self._registry[agent_id]
            return success

# Singleton instance
registry_manager = RegistryManager()

# Dependency for FastAPI
async def get_registry_manager():
    return registry_manager
```

## Fuzzy Matching Implementation
```python
from thefuzz import fuzz

def fuzzy_capability_match(agent_capabilities, requested_capabilities, threshold=70):
    """Match capabilities using fuzzy string matching with configurable threshold."""
    matches = []
    for req_cap in requested_capabilities:
        best_match_score = 0
        for agent_cap in agent_capabilities:
            score = fuzz.ratio(req_cap.lower(), agent_cap.lower())
            best_match_score = max(best_match_score, score)
        matches.append(best_match_score >= threshold)
    return all(matches)  # All requested capabilities must match
```

```python
# db/cache.py
import json
from typing import Any, Dict, List, Optional, Tuple, TypeVar, Generic
import redis.asyncio as redis
from fastapi import Depends

from core.config import settings

_redis_pool: Optional[redis.ConnectionPool] = None

async def init_redis_pool() -> redis.ConnectionPool:
    global _redis_pool
    if _redis_pool is None:
        _redis_pool = redis.ConnectionPool(
            host=settings.REDIS_HOST,
            port=settings.REDIS_PORT,
            db=settings.REDIS_DB,
            password=settings.REDIS_PASSWORD,
            ssl=settings.REDIS_SSL,
            decode_responses=True,
            max_connections=50
        )
    return _redis_pool

async def get_redis_client() -> redis.Redis:
    pool = await init_redis_pool()
    return redis.Redis(connection_pool=pool)

T = TypeVar('T')
class CacheManager(Generic[T]):
    """Generic cache manager for handling Redis caching operations"""
    
    def __init__(self, redis_client: redis.Redis, prefix: str, ttl_seconds: int):
        self.redis = redis_client
        self.prefix = prefix
        self.ttl = ttl_seconds
    
    def make_key(self, key_parts: Any) -> str:
        """Create a deterministic cache key from input parameters"""
        if isinstance(key_parts, dict):
            # Sort dict keys for consistent hashing
            serialized = json.dumps(key_parts, sort_keys=True)
        else:
            serialized = str(key_parts)
        
        import hashlib
        hashed = hashlib.md5(serialized.encode()).hexdigest()
        return f"{self.prefix}:{hashed}"
    
    async def get(self, key_parts: Any) -> Optional[T]:
        """Get item from cache"""
        key = self.make_key(key_parts)
        data = await self.redis.get(key)
        if data:
            return json.loads(data)
        return None
    
    async def set(self, key_parts: Any, value: T) -> None:
        """Set item in cache with TTL"""
        key = self.make_key(key_parts)
        await self.redis.setex(key, self.ttl, json.dumps(value))
    
    async def invalidate(self, key_parts: Optional[Any] = None) -> None:
        """Invalidate specific key or pattern"""
        if key_parts:
            key = self.make_key(key_parts)
            await self.redis.delete(key)
        else:
            # Invalidate all keys with this prefix
            pattern = f"{self.prefix}:*"
            keys = await self.redis.keys(pattern)
            if keys:
                await self.redis.delete(*keys)
```

```python
# services/cache_warmer.py
import asyncio
import logging
from datetime import datetime
from typing import List, Dict, Any

class CacheWarmer:

    def __init__(self, registry_manager: RegistryManager):
        self.registry = registry_manager
        
    async def warm_common_queries(self) -> None:
        """Warm cache with common discovery patterns"""
        logger.info("Starting cache warming process")
        
        # Common query patterns to pre-warm
        common_queries = [
            DiscoveryCriteria(),  # All agents
            DiscoveryCriteria(status="active"),  # Active agents
            DiscoveryCriteria(capabilities=["text-generation"]),  # Text generation agents
            DiscoveryCriteria(capabilities=["image-generation"]),  # Image generation agents
            DiscoveryCriteria(capabilities=["text-generation", "reasoning"]),  # LLM agents
        ]
        
        # Execute all queries to populate cache
        for criteria in common_queries:
            try:
                await self.registry.discover_agents(criteria)
                logger.debug(f"Warmed cache for query: {criteria}")
            except Exception as e:
                logger.error(f"Error warming cache for {criteria}: {str(e)}")
                
        logger.info("Cache warming completed")
    
    async def start_periodic_warming(self, interval_minutes: int = 30) -> None:
        """Start periodic cache warming in background"""
        while True:
            await self.warm_common_queries()
            await asyncio.sleep(interval_minutes * 60)
```

### Criteria Matching Algorithm
```python
async def matches_criteria(agent_data: dict, criteria: SubscriptionCriteria, event_type: str) -> bool:
    """Determine if an agent event matches subscription criteria"""
    # Check event type first (most efficient filter)
    if event_type not in criteria.event_types:
        return False
        
    # For deletion events, we only check event_type
    if event_type == "agent_deleted":
        return True
        
    # Check other criteria
    if criteria.agent_types and agent_data["agent_type"] not in criteria.agent_types:
        return False
        
    if criteria.capabilities:
        agent_capabilities = set(agent_data["capabilities"])
        if not any(cap in agent_capabilities for cap in criteria.capabilities):
            return False
            
    if criteria.name_contains and criteria.name_contains.lower() not in agent_data["name"].lower():
        return False
        
    if criteria.min_trust_score is not None and agent_data["trust_score"] < criteria.min_trust_score:
        return False
        
    return True
```

## Rate Limiting Implementation
```python
# app/middleware/rate_limit.py
from fastapi import Request, HTTPException
import time
import redis
from app.core.config import settings

class RateLimitMiddleware:
    def __init__(self, redis_client: redis.Redis):
        self.redis_client = redis_client
        self.rate_limit = settings.RATE_LIMIT_PER_MINUTE
        self.window = 60  # seconds
        
    async def __call__(self, request: Request, call_next):
        # Get client identifier (IP or user ID if authenticated)
        client_id = request.client.host
        if "user_id" in request.state:
            client_id = f"user:{request.state.user_id}"
            
        # Implement sliding window rate limiting
        current_time = int(time.time())
        key = f"ratelimit:{client_id}:{current_time // self.window}"
        
        # Increment counter for this window
        current_count = self.redis_client.incr(key)
        # Set expiry if this is a new key
        if current_count == 1:
            self.redis_client.expire(key, self.window * 2)
            
        if current_count > self.rate_limit:
            raise HTTPException(status_code=429, detail="Rate limit exceeded")
            
        response = await call_next(request)
        return response
```

```python
            health_status["status"] = "degraded"
    except Exception:
        health_status["components"]["cache"] = "error"
        health_status["status"] = "degraded"
    
    # Check subscription service
    from app.services.subscription_service import subscription_service
    if not subscription_service.is_running():
        health_status["components"]["subscription_service"] = "error"
        health_status["status"] = "degraded"
    
    return health_status
```

```python
class RegistryException(Exception):
    """Base exception for registry service"""
    def __init__(self, message: str, status_code: int = 400):
        self.message = message
        self.status_code = status_code
        super().__init__(self.message)

class AgentNotFoundException(RegistryException):
    """Raised when an agent is not found"""
    def __init__(self, agent_id: str):
        super().__init__(f"Agent with ID {agent_id} not found", 404)

class SubscriptionException(RegistryException):
    """Base exception for subscription operations"""
    pass

# app/error_handlers.py
from fastapi import Request, status
from fastapi.responses import JSONResponse
from app.exceptions import RegistryException, AgentNotFoundException, SubscriptionException

async def registry_exception_handler(request: Request, exc: RegistryException):
    return JSONResponse(
        status_code=exc.status_code,
        content={"detail": exc.message}
    )

def setup_error_handlers(app):
    app.add_exception_handler(RegistryException, registry_exception_handler)
    app.add_exception_handler(AgentNotFoundException, registry_exception_handler)
    app.add_exception_handler(SubscriptionException, registry_exception_handler)
```
