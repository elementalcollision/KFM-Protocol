from pydantic import BaseModel, Field

class OperatorType(str, Enum):
    EQUAL = "=="
    NOT_EQUAL = "!="
    GREATER_THAN = ">"
    GREATER_THAN_EQUAL = ">="
    LESS_THAN = "<"
    LESS_THAN_EQUAL = "<="
    CONTAINS = "contains"
    NOT_CONTAINS = "not_contains"
    REGEX_MATCH = "regex_match"

class ConditionType(str, Enum):
    TIME_IN_STATE = "time_in_state"
    METRIC_THRESHOLD = "metric_threshold"
    DEPENDENCY_CHECK = "dependency_check"
    METADATA_CHECK = "metadata_check"
    COMPOSITE = "composite"

class LogicalOperator(str, Enum):
    AND = "and"
    OR = "or"
    NOT = "not"

class ActionType(str, Enum):
    TRIGGER_KFM_OPERATION = "trigger_kfm_operation"
    NOTIFY = "notify"
    LOG = "log"
    WEBHOOK = "webhook"

class Condition(BaseModel):
    type: ConditionType
    
class MetricThresholdCondition(Condition):
    type: ConditionType = ConditionType.METRIC_THRESHOLD
    metric_name: str
    operator: OperatorType
    value: Union[int, float, str]

class TimeInStateCondition(Condition):
    type: ConditionType = ConditionType.TIME_IN_STATE
    operator: OperatorType
    value: str  # Duration string like "365 days"

class CompositeCondition(Condition):
    type: ConditionType = ConditionType.COMPOSITE
    operator: LogicalOperator
    conditions: List[Union["CompositeCondition", MetricThresholdCondition, TimeInStateCondition]]

class Action(BaseModel):
    type: ActionType
    
class PolicyRule(BaseModel):
    id: str
    name: str
    description: str
    enabled: bool = True
    target: Dict[str, Any]
    conditions: List[Condition]
    actions: List[Action]

# policy_engine/parser.py
import yaml
from typing import Dict, List, Union, Any
from .models import PolicyRule

def load_policy_rules(yaml_file_path: str) -> List[PolicyRule]:
    """
    Load and validate policy rules from a YAML file.
    
    Args:
        yaml_file_path: Path to the YAML file containing policy rules
        
    Returns:
        List of validated PolicyRule objects
        
    Raises:
        ValidationError: If the YAML content doesn't match the expected schema
        FileNotFoundError: If the YAML file doesn't exist
    """
    with open(yaml_file_path, 'r') as file:
        data = yaml.safe_load(file)
    
    if not isinstance(data, dict) or 'policies' not in data:
        raise ValueError("Invalid policy file format: missing 'policies' key")
    
    return [PolicyRule.parse_obj(policy) for policy in data['policies']]

def serialize_policy_rules(rules: List[PolicyRule], output_path: str) -> None:
    """
    Serialize policy rules to YAML and write to file.
    
    Args:
        rules: List of PolicyRule objects
        output_path: Path to write the YAML file
    """
    data = {'policies': [rule.dict() for rule in rules]}
    with open(output_path, 'w') as file:
        yaml.dump(data, file, default_flow_style=False, sort_keys=False)
```

logger = logging.getLogger("policy-validator")

def validate_policies(policies_dir):
    """Validates all policy files in the specified directory"""

            logger.info(f"✅ {policy_file} is valid")

            logger.error(f"❌ {policy_file} is invalid: {str(e)}")
            success = False

if __name__ == "__main__":

class ConditionEvaluator:
    """Strategy pattern implementation for different condition types."""
    
    def __init__(self, metric_service, dependency_service):
        self.metric_service = metric_service
        self.dependency_service = dependency_service
        self.evaluators = {
            'metric_threshold': self._evaluate_metric_threshold,
            'time_in_state': self._evaluate_time_in_state,
            'dependency_check': self._evaluate_dependency_check,
            'metadata_match': self._evaluate_metadata_match,
            # Extensibility point: register new condition types here
        }
    
    def evaluate(self, condition, agent_context):
        """Evaluate a condition against agent context."""
        if condition['type'] in ['AND', 'OR']:
            results = [self.evaluate(subcond, agent_context) 
                      for subcond in condition['conditions']]
            return all(results) if condition['type'] == 'AND' else any(results)
        
        elif condition['type'] == 'NOT':
            return not self.evaluate(condition['condition'], agent_context)
        
        elif condition['type'] in self.evaluators:
            return self.evaluators[condition['type']](condition, agent_context)
        
        else:
            raise ValueError(f"Unknown condition type: {condition['type']}")

# Performance optimization for the engine
class PolicyEvaluationCache:
    """Caches recent evaluation results to avoid redundant processing."""
    def __init__(self, ttl_seconds=60):
        self.cache = {}
        self.ttl_seconds = ttl_seconds
    
    def get(self, policy_id, agent_id, context_hash):
        """Retrieve cached result if valid and not expired."""
        key = (policy_id, agent_id, context_hash)
        if key in self.cache:
            result, timestamp = self.cache[key]
            if (datetime.now() - timestamp).total_seconds() < self.ttl_seconds:
                return result
        return None
    
    def set(self, policy_id, agent_id, context_hash, result):
        """Cache an evaluation result."""
        self.cache[(policy_id, agent_id, context_hash)] = (result, datetime.now())

async def _call_kfm_operator(self, operation_type: KFMOperationType, payload: dict) -> dict:
    """Make a call to a KFM operator with retry logic."""
    base_url = self._get_operator_url(operation_type)
    endpoint = KFM_OPERATOR_ENDPOINTS[operation_type]
    url = f"{base_url}{endpoint}"
    
    max_retries = 3
    retry_delay = 1.0  # seconds
    
    for attempt in range(max_retries):
        try:
            async with httpx.AsyncClient(timeout=10.0) as client:
                response = await client.post(url, json=payload)
                response.raise_for_status()
                return response.json()
        except (httpx.RequestError, httpx.HTTPStatusError) as e:
            logger.error(f"KFM operator call failed (attempt {attempt+1}/{max_retries}): {str(e)}")
            if attempt < max_retries - 1:
                await asyncio.sleep(retry_delay * (2 ** attempt))  # Exponential backoff
            else:
                raise KFMOperatorError(f"Failed to call {operation_type} operator after {max_retries} attempts") from e
```

def query_audit_logs(
    policy_id=None, 
    agent_id=None, 
    start_time=None, 
    end_time=None, 
    result=None, 
    limit=100
):
    """
    Basic implementation for querying audit logs from file
    Note: In production, this would likely use a database or log aggregation tool
    """
    matching_entries = []
    with open("/var/log/policy_engine/audit.log", "r") as f:
        for line in f:
            try:
                log_entry = json.loads(line)
                audit_data = log_entry.get("audit", {})
                
                # Apply filters
                if policy_id and audit_data.get("policy_id") != policy_id:
                    continue
                if agent_id and audit_data.get("agent_id") != agent_id:
                    continue
                if result is not None and audit_data.get("evaluation_result") != result:
                    continue
                if start_time and audit_data.get("timestamp", "") < start_time:
                    continue
                if end_time and audit_data.get("timestamp", "") > end_time:
                    continue
                
                matching_entries.append(audit_data)
                if len(matching_entries) >= limit:
                    break
            except json.JSONDecodeError:
                continue
                
    return matching_entries

import os
import logging
import json
from datetime import datetime
from pathlib import Path

class JsonFormatter(logging.Formatter):
    """Custom formatter that outputs log records as JSON objects"""
    def format(self, record):
        log_data = {
            "message": record.getMessage(),
            "level": record.levelname,
            "timestamp": datetime.utcnow().isoformat(),
            "logger": record.name,
            "module": record.module,
            "function": record.funcName,
            "line": record.lineno
        }
        
        # Include any exception information
        if record.exc_info:
            log_data["exception"] = {
                "type": record.exc_info[0].__name__,
                "message": str(record.exc_info[1]),
                "traceback": self.formatException(record.exc_info)
            }
            
        # Include audit data if present
        if hasattr(record, 'audit_data'):
            log_data["audit"] = record.audit_data
            
        return json.dumps(log_data)

def configure_policy_audit_logging(log_file_path="logs/policy_audit.log", 
                                  log_level=logging.INFO,
                                  rotation_size_mb=10,
                                  backup_count=5):
    """
    Configure the policy audit logging system
    
    Args:
        log_file_path: Path to the log file
        log_level: Logging level (default: INFO)
        rotation_size_mb: Size in MB before rotating log file
        backup_count: Number of backup files to keep
    
    Returns:
        The configured logger
    """
    # Create directory if it doesn't exist
    log_dir = os.path.dirname(log_file_path)
    Path(log_dir).mkdir(parents=True, exist_ok=True)
    
    # Get logger
    audit_logger = logging.getLogger("policy_audit")
    audit_logger.setLevel(log_level)
    
    # Remove existing handlers if any
    for handler in audit_logger.handlers[:]:
        audit_logger.removeHandler(handler)
    
    # Create rotating file handler
    from logging.handlers import RotatingFileHandler
    file_handler = RotatingFileHandler(
        log_file_path,
        maxBytes=rotation_size_mb * 1024 * 1024,
        backupCount=backup_count
    )
    
    # Set formatter
    file_handler.setFormatter(JsonFormatter())
    audit_logger.addHandler(file_handler)
    
    # Optionally add console handler for development environments
    if os.getenv("POLICY_ENGINE_ENV") == "development":
        console_handler = logging.StreamHandler()
        console_handler.setFormatter(JsonFormatter())
        audit_logger.addHandler(console_handler)
    
    audit_logger.info("Policy audit logging initialized")
    return audit_logger

   ```python
   def reload_policies(self, policy_file_path: str):
       """Atomically reload policies with validation and rollback."""
       # Store current policies for potential rollback
       previous_policies = copy.deepcopy(self.policies)
       
       try:
           # Load and validate new policies
           new_policies = self.load_and_validate_policies(policy_file_path)
           
           # Atomic replacement
           with self._policy_lock:  # Thread safety
               self.policies = new_policies
               
           # Signal successful reload to listeners
           self._notify_policy_change()
           return True
       except Exception as e:
           # Rollback to previous policies
           self.policies = previous_policies
           logger.error(f"Policy reload failed, rolled back: {str(e)}")
           raise
   ```
