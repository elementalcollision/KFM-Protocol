# app/core/security.py
from fastapi import Depends, HTTPException, Security
from fastapi.security import APIKeyHeader, OAuth2PasswordBearer
from jose import JWTError, jwt
from datetime import datetime, timedelta
from typing import Optional, Dict, Any

class SecurityManager:
    def __init__(self, config):
        self.config = config
        self.oauth2_scheme = OAuth2PasswordBearer(tokenUrl="token")
        self.api_key_header = APIKeyHeader(name="X-API-Key")
        
    async def authenticate_request(self, token: str = Depends(oauth2_scheme)):
        """Validate JWT token and extract claims"""
        try:
            payload = jwt.decode(
                token, 
                self.config.SECRET_KEY, 
                algorithms=[self.config.ALGORITHM]
            )
            return payload
        except JWTError:
            raise HTTPException(status_code=401, detail="Invalid authentication credentials")
    
    async def authorize_service_access(self, 
                                      service_name: str,
                                      token_data: Dict[str, Any] = Depends(authenticate_request)):
        """Check if the authenticated client has access to the requested service"""
        if "permissions" not in token_data:
            raise HTTPException(status_code=403, detail="No permissions found")
            
        permissions = token_data["permissions"]
        if f"service:{service_name}:access" not in permissions:
            raise HTTPException(status_code=403, detail=f"Not authorized to access {service_name}")
        
        return token_data

## Rate Limiting Implementation

```python
# app/core/rate_limiter.py
import time
import asyncio
from fastapi import Request, HTTPException
from typing import Dict, Tuple

class RateLimiter:
    def __init__(self, rate_limit: int, time_window: int):
        """
        Initialize rate limiter
        
        Args:
            rate_limit: Maximum number of requests allowed
            time_window: Time window in seconds
        """
        self.rate_limit = rate_limit
        self.time_window = time_window
        self.clients: Dict[str, Tuple[int, float]] = {}  # client_id -> (request_count, window_start)
        
    async def check_rate_limit(self, request: Request):
        """Check if request exceeds rate limit"""
        # Extract client identifier (IP or token subject)
        client_id = self._get_client_id(request)
        
        current_time = time.time()
        if client_id in self.clients:
            count, window_start = self.clients[client_id]
            
            # Reset if window has expired
            if current_time - window_start > self.time_window:
                self.clients[client_id] = (1, current_time)
            else:
                # Increment count if within window
                if count >= self.rate_limit:
                    raise HTTPException(
                        status_code=429, 
                        detail="Rate limit exceeded. Try again later."
                    )
                self.clients[client_id] = (count + 1, window_start)
        else:
            # First request from this client
            self.clients[client_id] = (1, current_time)
        
        return True
        
    def _get_client_id(self, request: Request) -> str:
        """Extract client identifier from request"""
        # Use token subject if available, otherwise use IP
        if hasattr(request.state, "user") and "sub" in request.state.user:
            return request.state.user["sub"]
        return request.client.host
```

# app/middleware/gateway_middleware.py
from fastapi import Request, Response
from starlette.middleware.base import BaseHTTPMiddleware
import time
import logging
from app.core.rate_limiter import RateLimiter
from app.core.metrics import MetricsCollector

logger = logging.getLogger("api_gateway")

class GatewayMiddleware(BaseHTTPMiddleware):
    def __init__(self, app, rate_limiter: RateLimiter, metrics: MetricsCollector):
        super().__init__(app)
        self.rate_limiter = rate_limiter
        self.metrics = metrics
        
    async def dispatch(self, request: Request, call_next):
        # Start timer for request duration
        start_time = time.time()
        
        # Check rate limits
        await self.rate_limiter.check_rate_limit(request)
        
        # Extract request ID for correlation
        request_id = self._get_request_id(request)
        
        # Add correlation ID to request state
        request.state.request_id = request_id
        
        # Log incoming request
        logger.info(f"Request {request_id}: {request.method} {request.url.path}")
        
        try:
            # Process the request
            response = await call_next(request)
            
            # Record metrics
            self.metrics.record_request(
                service=self._extract_service_name(request),
                status_code=response.status_code,
                duration=time.time() - start_time
            )
            
            # Add correlation ID to response headers
            response.headers["X-Request-ID"] = request_id
            
            return response
            
        except Exception as e:
            # Log exception
            logger.exception(f"Error processing request {request_id}: {str(e)}")
            
            # Record error metric
            self.metrics.record_error(
                service=self._extract_service_name(request),
                error_type=type(e).__name__
            )
            
            # Re-raise to let exception handlers deal with it
            raise
            
    def _get_request_id(self, request: Request) -> str:
        """Extract request ID from headers or generate new one"""
        if "X-Request-ID" in request.headers:
            return request.headers["X-Request-ID"]
        return str(uuid.uuid4())
        
    def _extract_service_name(self, request: Request) -> str:
        """Extract service name from request path"""
        path_parts = request.url.path.split("/")
        if len(path_parts) >= 4 and path_parts[1] == "api":
            return path_parts[3]
        return "unknown"

## Circuit Breaker Pattern Implementation

```python
# app/core/circuit_breaker.py
import time
from enum import Enum
from typing import Callable, Any, Dict
import asyncio
import logging

logger = logging.getLogger("circuit_breaker")

class CircuitState(Enum):
    CLOSED = "CLOSED"  # Normal operation
    OPEN = "OPEN"      # Failing, rejecting requests
    HALF_OPEN = "HALF_OPEN"  # Testing if service recovered

class CircuitBreaker:
    def __init__(self, 
                failure_threshold: int = 5,
                recovery_timeout: int = 30,
                timeout_duration: float = 10.0):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.timeout_duration = timeout_duration
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.last_failure_time = 0
        self.services: Dict[str, CircuitState] = {}
        
    async def execute(self, service_name: str, func: Callable, *args, **kwargs) -> Any:
        """Execute function with circuit breaker protection"""
        # Check if circuit is OPEN
        if self._get_service_state(service_name) == CircuitState.OPEN:
            if time.time() - self._get_last_failure_time(service_name) > self.recovery_timeout:
                # Try recovery
                self._set_service_state(service_name, CircuitState.HALF_OPEN)
                logger.info(f"Circuit for {service_name} changed to HALF_OPEN")
            else:
                # Circuit is open, reject request
                raise CircuitOpenError(f"Circuit for {service_name} is OPEN")
                
        try:
            # Set timeout for the function call
            result = await asyncio.wait_for(func(*args, **kwargs), timeout=self.timeout_duration)
            
            # Success - reset if in HALF_OPEN
            if self._get_service_state(service_name) == CircuitState.HALF_OPEN:
                self._reset_service(service_name)
                logger.info(f"Circuit for {service_name} recovered and CLOSED")
                
            return result
            
        except (Exception, asyncio.TimeoutError) as e:
            # Failure - increment counter
            self._increment_failure(service_name)
            
            # Check if threshold reached
            if (self._get_service_state(service_name) == CircuitState.CLOSED and 
                self._get_failure_count(service_name) >= self.failure_threshold):
                self._set_service_state(service_name, CircuitState.OPEN)
                self._set_last_failure_time(service_name, time.time())
                logger.warning(f"Circuit for {service_name} changed to OPEN after {self.failure_threshold} failures")
                
            # If HALF_OPEN, go back to OPEN
            elif self._get_service_state(service_name) == CircuitState.HALF_OPEN:
                self._set_service_state(service_name, CircuitState.OPEN)
                self._set_last_failure_time(service_name, time.time())
                logger.warning(f"Circuit for {service_name} changed back to OPEN after failed recovery attempt")
                
            # Re-raise the original exception
            raise
            
    # Service state management methods
    def _get_service_state(self, service_name: str) -> CircuitState:
        if service_name not in self.services:
            self.services[service_name] = {
                "state": CircuitState.CLOSED,
                "failure_count": 0,
                "last_failure_time": 0
            }
        return self.services[service_name]["state"]
        
    def _set_service_state(self, service_name: str, state: CircuitState):
        if service_name not in self.services:
            self.services[service_name] = {
                "state": state,
                "failure_count": 0,
                "last_failure_time": 0
            }
        else:
            self.services[service_name]["state"] = state
            
    # Additional helper methods for service state management
    def _get_failure_count(self, service_name: str) -> int:
        return self.services.get(service_name, {}).get("failure_count", 0)
        
    def _increment_failure(self, service_name: str):
        if service_name not in self.services:
            self.services[service_name] = {
                "state": CircuitState.CLOSED,
                "failure_count": 1,
                "last_failure_time": time.time()
            }
        else:
            self.services[service_name]["failure_count"] += 1
            
    def _get_last_failure_time(self, service_name: str) -> float:
        return self.services.get(service_name, {}).get("last_failure_time", 0)
        
    def _set_last_failure_time(self, service_name: str, time_value: float):
        if service_name in self.services:
            self.services[service_name]["last_failure_time"] = time_value
            
    def _reset_service(self, service_name: str):
        self.services[service_name] = {
            "state": CircuitState.CLOSED,
            "failure_count": 0,
            "last_failure_time": 0
        }

class CircuitOpenError(Exception):
    """Exception raised when circuit is open"""
    pass
```

@router.api_route("/{service}/{path:path}", methods=["GET", "POST", "PUT", "DELETE"])
async def dynamic_route(
    request: Request,
    service: str,
    path: str,
    service_registry: ServiceRegistry = Depends(get_service_registry),
    http_client: HTTPClientService = Depends(get_http_client)
):
    try:
        # Get target service URL
        service_url = await service_registry.get_service_url(service)
        
        # Extract path parameters
        path_params = request.path_params.copy()
        path_params.pop("service")
        path_params.pop("path")
        
        # Forward request to backend service
        response = await http_client.forward_request(
            method=request.method,
            url=f"{service_url}/{path}",
            headers=dict(request.headers),
            params=dict(request.query_params),
            path_params=path_params,
            body=await request.body() if request.method in ["POST", "PUT", "PATCH"] else None
        )
        
        return response
    except ServiceNotFoundError as e:
        return JSONResponse(
            status_code=404,
            content={"error": str(e), "code": "SERVICE_NOT_FOUND"}
        )
    except ServiceUnavailableError as e:
        return JSONResponse(

            url = url.replace(f"{{{param}}}", value)
            
        # Configure retry with exponential backoff
        retry_options = {

            method=method,
            url=url,
            headers=headers,
            params=params,
            content=body,

    async def delete(self, url: str, params: Optional[Dict] = None, headers: Optional[Dict] = None):
        """Execute DELETE request with retry and error handling"""
        return await self._execute_with_retry("DELETE", url, params=params, headers=headers)
        
    async def _execute_with_retry(self, method: str, url: str, **kwargs):
        """Execute request with retry logic"""
        retries = 0
        last_exception = None
        
        while retries <= self.max_retries:
            try:
                response = await self.client.request(method, url, **kwargs)
                
                # Check if status code indicates we should retry
                if response.status_code in self.retry_status_codes and retries < self.max_retries:
                    retries += 1
                    await asyncio.sleep(self.retry_backoff * (2 ** retries))
                    continue
                    
                # Parse response based on content type
                if response.status_code >= 400:
                    return self._handle_error_response(response)
                    
                return response.json() if response.headers.get("content-type") == "application/json" else response.text
                
            except (httpx.RequestError, httpx.TimeoutException) as e:
                last_exception = e
                retries += 1
                
                if retries <= self.max_retries:
                    await asyncio.sleep(self.retry_backoff * (2 ** retries))
                else:
                    break
                    
        # If we've exhausted retries, raise the last exception
        raise ServiceConnectionError(f"Failed to connect to service after {self.max_retries} retries: {str(last_exception)}")

        self.failure_threshold = failure_threshold
        self.recovery_time = recovery_time
        self.timeout = timeout
        self.failure_count = 0
        self.last_failure_time = 0
        self.state = "CLOSED"  # CLOSED, OPEN, HALF-OPEN
        self.lock = asyncio.Lock()
        
    async def execute(self, func, *args, **kwargs):
        async with self.lock:
            if self.state == "OPEN":
                if time.time() - self.last_failure_time > self.recovery_time:
                    self.state = "HALF-OPEN"
                else:
                    raise CircuitBreakerOpenError("Circuit breaker is open")
                    
        try:
            result = await asyncio.wait_for(func(*args, **kwargs), timeout=self.timeout)
            
            if self.state == "HALF-OPEN":
                async with self.lock:
                    self.state = "CLOSED"
                    self.failure_count = 0
                    
            return result
            
        except (asyncio.TimeoutError, Exception) as e:
            async with self.lock:
                self.failure_count += 1
                self.last_failure_time = time.time()
                
                if self.failure_count >= self.failure_threshold or self.state == "HALF-OPEN":
                    self.state = "OPEN"
                    
            if isinstance(e, asyncio.TimeoutError):
                raise ServiceTimeoutError("Service request timed out")
            raise e

# circuit_breaker.py
import time
from enum import Enum

class CircuitState(Enum):
    CLOSED = "closed"  # Normal operation
    OPEN = "open"      # Failing, rejecting requests
    HALF_OPEN = "half_open"  # Testing if service recovered

class CircuitBreaker:
    def __init__(self, failure_threshold=5, recovery_timeout=30, half_open_max_calls=3):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.half_open_max_calls = half_open_max_calls
        
        self.state = CircuitState.CLOSED
        self.failure_count = 0
        self.last_failure_time = 0
        self.half_open_calls = 0
    
    async def execute(self, func, *args, **kwargs):
        if self.state == CircuitState.OPEN:
            # Check if recovery timeout has elapsed
            if time.time() - self.last_failure_time > self.recovery_timeout:
                self.state = CircuitState.HALF_OPEN
                self.half_open_calls = 0
            else:
                raise CircuitBreakerOpenException("Circuit breaker is open")
        
        if self.state == CircuitState.HALF_OPEN and self.half_open_calls >= self.half_open_max_calls:
            raise CircuitBreakerOpenException("Circuit breaker is half-open and at capacity")
        
        try:
            if self.state == CircuitState.HALF_OPEN:
                self.half_open_calls += 1
                
            result = await func(*args, **kwargs)
            
            # Success - reset if in half-open state
            if self.state == CircuitState.HALF_OPEN:
                self.state = CircuitState.CLOSED
                self.failure_count = 0
            
            return result
            
        except Exception as e:
            self.failure_count += 1
            self.last_failure_time = time.time()
            
            if self.state == CircuitState.HALF_OPEN or self.failure_count >= self.failure_threshold:
                self.state = CircuitState.OPEN
            
            raise e

           # Fallback to cached data if available
