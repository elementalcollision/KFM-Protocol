### Template Selection Algorithm
```python
def get_template_for_promotion(self, entity_id: int, entity_type: str, 
                              from_level: int, to_level: int, department_id: Optional[int] = None) -> PromotionChecklistTemplate:
    """
    Select the most specific template matching the promotion criteria.
    Templates are matched in order of specificity:
    1. Exact entity_type, from_level, to_level, and department_id
    2. Exact entity_type, from_level, to_level, any department
    3. Exact entity_type, any from_level to to_level
    4. Default template for entity_type
    """
    query = self.db.query(PromotionChecklistTemplate).filter(
        PromotionChecklistTemplate.entity_type == entity_type,
        PromotionChecklistTemplate.is_active == True
    )
    
    # Try with specific department and levels
    template = query.filter(
        PromotionChecklistTemplate.from_level == from_level,
        PromotionChecklistTemplate.to_level == to_level,
        PromotionChecklistTemplate.department_id == department_id
    ).first()
    
    if not template:
        # Try with specific levels but any department
        template = query.filter(
            PromotionChecklistTemplate.from_level == from_level,
            PromotionChecklistTemplate.to_level == to_level,
            PromotionChecklistTemplate.department_id == None
        ).first()
    
    if not template:
        # Try default template for entity type
        template = query.filter(
            PromotionChecklistTemplate.from_level == 0,
            PromotionChecklistTemplate.to_level == 0,
            PromotionChecklistTemplate.department_id == None
        ).first()
        
    if not template:
        raise TemplateNotFoundException(f"No template found for {entity_type} promotion from level {from_level} to {to_level}")
        
    return template
```

### Checklist Generation Implementation
```python
def generate_checklist_items(self, review_id: int, template_id: int) -> List[PromotionChecklistItem]:
    """
    Generate checklist items for a promotion review based on a template
    """
    template = self.db.query(PromotionChecklistTemplate).get(template_id)
    if not template:
        raise TemplateNotFoundException(f"Template with ID {template_id} not found")
        
    items = []
    for criteria in template.criteria:
        item = PromotionChecklistItem(
            review_id=review_id,
            criteria_id=criteria.id,
            name=criteria.name,
            description=criteria.description,
            required=criteria.required,
            status="pending",
            order=criteria.order
        )
        items.append(item)
        
    self.db.add_all(items)
    self.db.commit()
    return items
```

## Approval Status Transition Logic
```python
async def process_approval_response(
    review_id: UUID, 
    stakeholder_id: UUID, 
    status: str, 
    notes: Optional[str], 
    db: Session
) -> Dict:
    """Process an approval response and update review status if needed"""
    
    # Update the specific approval
    approval = await crud.get_pending_approval(db, review_id, stakeholder_id)
    if not approval:
        raise HTTPException(404, "No pending approval found for this stakeholder")
    
    approval_update = {
        "status": status,
        "notes": notes,
        "responded_at": datetime.now()
    }
    
    updated_approval = await crud.update_approval_status(db, approval, approval_update)
    
    # Check if this completes or fails the review
    review = await crud.get_promotion_review(db, review_id)
    workflow_config = settings.approval_workflows.get(review.workflow_type, "default")
    
    # Get all approvals for this review
    all_approvals = await crud.get_approvals_by_review(db, review_id)
    
    # If any required role rejects, fail the review
    if status == "REJECTED":
        await crud.update_promotion_review(
            db, 
            review, 
            {"status": "REJECTED", "rejection_reason": f"Rejected by {approval.stakeholder.role}: {notes}"}
        )
        await audit_service.log_event(
            "promotion_rejected",
            f"Promotion rejected by {approval.stakeholder.full_name}",
            {"review_id": str(review_id), "stakeholder_id": str(stakeholder_id), "notes": notes}
        )
        return {"status": "REJECTED", "message": "Review rejected"}
    
    # Check if all required approvals are complete
    required_roles = [r["role"] for r in workflow_config["roles"] if r.get("required", True)]
    approved_roles = [a.role for a in all_approvals if a.status == "APPROVED"]
    
    # If threshold is defined, check if we've met it
    if "threshold" in workflow_config:
        approved_count = len([a for a in all_approvals if a.status == "APPROVED"])
        if approved_count >= workflow_config["threshold"]:
            await crud.update_promotion_review(db, review, {"status": "APPROVED"})
            await trigger_promotion_execution(review, db)
            return {"status": "APPROVED", "message": "Review approved, promotion initiated"}
    
    # Otherwise check if all required roles have approved
    elif all(role in approved_roles for role in required_roles):
        await crud.update_promotion_review(db, review, {"status": "APPROVED"})
        await trigger_promotion_execution(review, db)
        return {"status": "APPROVED", "message": "Review approved, promotion initiated"}
    
    # Still waiting for more approvals
    pending_roles = [r for r in required_roles if r not in approved_roles]
    return {
        "status": "PENDING",
        "message": f"Approval recorded. Still waiting for: {', '.join(pending_roles)}"
    }
```

## Audit Trail Implementation
```python
class ApprovalAuditContext:
    """Context manager for approval actions to ensure proper audit logging"""
    
    def __init__(self, db: Session, review_id: UUID, stakeholder_id: UUID, action: str):
        self.db = db
        self.review_id = review_id
        self.stakeholder_id = stakeholder_id
        self.action = action
        self.start_time = None
        
    async def __aenter__(self):
        self.start_time = datetime.now()
        await audit_service.log_event(
            f"approval_{self.action}_started",
            f"Approval {self.action} initiated by stakeholder {self.stakeholder_id}",
            {"review_id": str(self.review_id)}
        )
        return self
        
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        duration = (datetime.now() - self.start_time).total_seconds()
        
        if exc_type:
            await audit_service.log_event(
                f"approval_{self.action}_failed",
                f"Approval {self.action} failed: {str(exc_val)}",
                {
                    "review_id": str(self.review_id),
                    "stakeholder_id": str(self.stakeholder_id),
                    "duration_seconds": duration,
                    "error": str(exc_val)
                }
            )
        else:
            await audit_service.log_event(
                f"approval_{self.action}_completed",
                f"Approval {self.action} completed successfully",
                {
                    "review_id": str(self.review_id),
                    "stakeholder_id": str(self.stakeholder_id),
                    "duration_seconds": duration
                }
            )
```

```python
def check_for_due_reviews(db: Session):
    """Daily job to check for reviews that are due or overdue"""
    today = datetime.utcnow().date()
    
    # Get reviews due today or already overdue
    due_reviews = crud_maintenance.get_due_or_overdue_reviews(db)
    
    for review in due_reviews:
        if review.status == ReviewStatus.SCHEDULED:
            if review.due_date.date() < today:
                # Mark as overdue
                crud_maintenance.update_maintenance_review(
                    db, 
                    review.id, 
                    {"status": ReviewStatus.OVERDUE}
                )
                # Send escalation notification
                notification_service.send_overdue_review_notification(review)
            else:
                # Send due notification
                notification_service.send_due_review_notification(review)

def complete_review(
    db: Session,
    review_id: UUID,
    outcome: ReviewOutcome,
    notes: str = None,
    reviewer_id: UUID = None,
    schedule_next: bool = True
) -> MaintenanceReview:
    """Complete a review with outcome and optionally schedule the next one"""
    
    update_data = {
        "status": ReviewStatus.COMPLETED,
        "completed_date": datetime.utcnow(),
        "outcome": outcome,
        "notes": notes,
        "reviewer_id": reviewer_id
    }
    
    updated_review = crud_maintenance.update_maintenance_review(db, review_id, update_data)
    
    # Schedule next review if needed
    if schedule_next and outcome != ReviewOutcome.ARCHIVED:
        next_interval = _get_next_review_interval(db, updated_review)
        next_review = schedule_next_review(
            db, 
            updated_review.agent_id,
            updated_review.review_type,
            days_until_review=next_interval
        )
        
        # Update the current review with next_review_date
        crud_maintenance.update_maintenance_review(
            db,
            review_id,
            {"next_review_date": next_review.scheduled_date}
        )
    
    return updated_review

def _get_next_review_interval(db: Session, review: MaintenanceReview) -> int:
    """Determine the appropriate interval for the next review based on outcome and history"""
    base_interval = settings.MAINTENANCE_REVIEW_INTERVAL_DAYS
    
    # Adjust interval based on outcome
    if review.outcome == ReviewOutcome.NEEDS_CHANGES:
        # Schedule sooner if changes needed
        return min(base_interval, 30)  # Max 30 days
    elif review.outcome == ReviewOutcome.APPROVED:
        # Can use standard interval
        return base_interval
    
    return base_interval
```
