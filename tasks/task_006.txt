```python
from abc import ABC, abstractmethod

from app.core.config import Settings
from app.schemas.adaptation import AdaptationRequest, AdaptationResponse

class AdaptationInterface(ABC):
    """Interface for adaptation services."""
    
    def __init__(self, settings: Settings):
        self.settings = settings
        self.logger = logging.getLogger(f"{__name__}.{self.__class__.__name__}")
    
    @abstractmethod
    async def adapt(self, agent_id: str, request: AdaptationRequest) -> AdaptationResponse:
        """Perform adaptation for the specified agent."""
        pass
    
    @abstractmethod
    async def validate_request(self, agent_id: str, request: AdaptationRequest) -> bool:
        """Validate the adaptation request."""
        pass
    
    async def log_adaptation_attempt(self, agent_id: str, request: AdaptationRequest, result: Optional[Dict[str, Any]] = None):
        """Log adaptation attempt for auditing and monitoring."""
        self.logger.info(
            "Adaptation attempt",
            extra={
                "agent_id": agent_id,
                "adaptation_type": request.adaptation_type,
                "parameters": request.parameters,
```

```python
# services/feature_flag_service.py
class FeatureFlagService:
    def __init__(self, config):
        self.flags = config.feature_flags
        self.dynamic_flags = {}  # Runtime-updated flags
        
    async def is_flag_active(self, flag_name: str, context: dict = None) -> bool:
        # Check dynamic flags first (experiment overrides)
        if flag_name in self.dynamic_flags:
            return self._evaluate_flag(self.dynamic_flags[flag_name], context)
            
        # Fall back to config flags
        if flag_name in self.flags:
            return self._evaluate_flag(self.flags[flag_name], context)
            
        return False
        
    def _evaluate_flag(self, flag_config, context):
        # Simple version - just check is_active
        if isinstance(flag_config, bool):
            return flag_config
            
        # Advanced version with targeting rules
        if isinstance(flag_config, dict):
            if "is_active" in flag_config and not flag_config["is_active"]:
                return False
                
            # Targeting logic (user percentage, environment, etc)
            if context and "targeting" in flag_config:
                return self._evaluate_targeting(flag_config["targeting"], context)
                
            return flag_config.get("is_active", False)
            
    def _evaluate_targeting(self, targeting, context):
        # Percentage rollout
        if "percentage" in targeting:
            user_id = context.get("user_id", "")
            if user_id:
                # Deterministic hashing for consistent user experience
                import hashlib
                hash_val = int(hashlib.md5(f"{user_id}:{targeting.get('seed', 'default')}".encode()).hexdigest(), 16)
                return (hash_val % 100) < targeting["percentage"]
        
        # Environment targeting
        if "environments" in targeting and "environment" in context:
            return context["environment"] in targeting["environments"]
            
        return True
```

```python
# services/metrics_service.py
class ExperimentMetricsService:
    """Service for collecting and analyzing experiment metrics"""
    
    def __init__(self, metrics_client, logger):
        self.metrics_client = metrics_client  # e.g., Prometheus, custom metrics store
        self.logger = logger
        
    async def collect_metrics(self, experiment, time_range=None):
        """Collect metrics for an experiment"""
        # Default to experiment duration if time range not specified
        if not time_range and experiment.start_date:
            start = experiment.start_date
            end = experiment.end_date or datetime.now()
            time_range = (start, end)
            
        metrics = {}
        
        # Collect metrics for each success criterion
        for criterion in experiment.success_criteria:
            metric_name = criterion["metric"]
            try:
                # Query metrics with experiment and variant labels
                control_data = await self.metrics_client.query(
                    metric_name,
                    {"experiment": experiment.name, "variant": "control"},
                    time_range
                )
                
                treatment_data = await self.metrics_client.query(
                    metric_name,
                    {"experiment": experiment.name, "variant": "treatment"},
                    time_range
                )
                
                metrics[metric_name] = {
                    "control": self._summarize_metric(control_data),
                    "treatment": self._summarize_metric(treatment_data)
                }
            except Exception as e:
                self.logger.error(f"Failed to collect metric {metric_name}: {str(e)}")
                
        return metrics
        
    def _summarize_metric(self, data):
        """Summarize metric data with statistical measures"""
        import numpy as np
        
        if not data:
            return {"count": 0}
            
        values = [point["value"] for point in data]
        return {
            "count": len(values),
            "mean": np.mean(values),
            "median": np.median(values),
            "p95": np.percentile(values, 95),
            "min": np.min(values),
            "max": np.max(values)
        }
        
    def analyze_results(self, experiment, metrics):
        """Analyze experiment results against success criteria"""
        results = {}
        
        for criterion in experiment.success_criteria:
            metric_name = criterion["metric"]
            if metric_name not in metrics:
                results[metric_name] = {"status": "missing_data"}
                continue
                
            metric_data = metrics[metric_name]
            
            # Calculate improvement
            control = metric_data["control"]["mean"]
            treatment = metric_data["treatment"]["mean"]
            
            if control == 0:
                relative_change = float('inf') if treatment > 0 else 0
            else:
                relative_change = (treatment - control) / control
                
            # Determine if criterion was met
            threshold = criterion.get("threshold", 0)
            direction = criterion.get("direction", "increase")
            
            if direction == "increase":
                success = relative_change >= threshold
            else:
                success = relative_change <= threshold
                
            results[metric_name] = {
                "control": control,
                "treatment": treatment,
                "relative_change": relative_change,
                "threshold": threshold,
                "direction": direction,
                "success": success
            }
            
        return results
```

```python
# clients/feature_flag_client.py
class FeatureFlagClient:
    """Client for components to check feature flags"""
    
    def __init__(self, feature_flag_service_url, auth_token=None):
        self.base_url = feature_flag_service_url
        self.headers = {"Authorization": f"Bearer {auth_token}"} if auth_token else {}
        self.cache = {}  # Simple cache to reduce API calls
        self.cache_ttl = 30  # seconds
        
    async def is_flag_active(self, flag_name: str, context: dict = None) -> bool:
        """Check if a feature flag is active"""
        cache_key = f"{flag_name}:{json.dumps(context or {})}"
        
        # Check cache first
        if cache_key in self.cache:
            entry = self.cache[cache_key]
            if time.time() - entry["timestamp"] < self.cache_ttl:
                return entry["value"]
        
        # Make API call
        async with aiohttp.ClientSession() as session:
            url = f"{self.base_url}/api/v1/feature-flags/evaluate/{flag_name}"
            async with session.post(url, json=context or {}, headers=self.headers) as response:
                if response.status == 200:
                    result = await response.json()
                    is_active = result.get("is_active", False)
                    
                    # Update cache
                    self.cache[cache_key] = {
                        "value": is_active,
                        "timestamp": time.time()
                    }
                    
                    return is_active
                else:
                    # Log error and default to False
                    error_text = await response.text()
                    logging.error(f"Error checking feature flag {flag_name}: {error_text}")
                    return False
```

### Lineage Graph Query API
```python
@app.get("/api/provenance/lineage/{entity_id}")
def get_entity_lineage(entity_id: UUID, depth: int = 5):
    """
    Returns the complete lineage graph for an entity up to specified depth.
    
    The returned data structure is optimized for visualization with libraries 
    like D3.js or Cytoscape.js.
    """
    nodes = []
    edges = []
    
    # Start with the requested entity
    entity_queue = [(get_entity(entity_id), 0)]
    processed_entities = set()
    
    while entity_queue:
        current_entity, current_depth = entity_queue.pop(0)
        
        if current_entity.id in processed_entities or current_depth > depth:
            continue
            
        processed_entities.add(current_entity.id)
        
        # Add node for current entity
        nodes.append({
            "id": str(current_entity.id),
            "label": current_entity.name,
            "type": current_entity.type,
            "stability": current_entity.stability_status
        })
        
        # Get parent entities from provenance records
        parent_records = get_provenance_records(
            entity_id=current_entity.id, 
            include_parents=True
        )
        
        for record in parent_records:
            if record.parent_entity_id:
                # Add edge from parent to current
                edges.append({
                    "source": str(record.parent_entity_id),
                    "target": str(current_entity.id),
                    "operation": record.operation_type,
                    "timestamp": record.timestamp.isoformat()
                })
                
                # Add parent to queue for processing
                parent_entity = get_entity(record.parent_entity_id)
                entity_queue.append((parent_entity, current_depth + 1))
    
    return {"nodes": nodes, "edges": edges}
```
