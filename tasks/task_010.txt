2. **Logstash Pipeline Configuration:**
   ```
   # logstash.conf
   input {
     beats {
       port => 5044
     }
   }
   
   filter {
     if [kubernetes][namespace] =~ /^kfm-/ {
       json {
         source => "message"
         target => "log_data"
         skip_on_invalid_json => true
       }
       
       # Extract common fields
       mutate {
         add_field => {
           "service_name" => "%{[kubernetes][container][name]}"
           "environment" => "%{[kubernetes][labels][environment]}"
         }
       }
       
       # Add correlation ID for request tracing
       if [log_data][request_id] {
         mutate {
           add_field => { "correlation_id" => "%{[log_data][request_id]}" }
         }
       }
       
       # Classify log levels for better filtering
       if [log_data][level] {
         mutate {
           add_field => { "log_level" => "%{[log_data][level]}" }
         }
       }
     }
   }
   
   output {
     elasticsearch {
       hosts => ["elasticsearch-service:9200"]
       index => "kfm-%{[kubernetes][namespace]}-%{[kubernetes][container][name]}-%{+YYYY.MM.dd}"
       template_name => "kfm-logs"
       template_overwrite => true
     }
   }
   ```

2. **Configure Tracing:**
   * Create `core/tracing_config.py` with:
     ```python
     from opentelemetry import trace
     from opentelemetry.sdk.trace import TracerProvider
     from opentelemetry.sdk.trace.export import BatchSpanProcessor
     from opentelemetry.sdk.resources import Resource
     from opentelemetry.exporter.otlp.proto.http.trace_exporter import OTLPSpanExporter
     from opentelemetry.trace.propagation.tracecontext import TraceContextTextMapPropagator
     from opentelemetry.context.context import Context
     from app.core.config import settings
     
     def configure_tracing():
         """Configure OpenTelemetry tracing for the service"""
         # Create resource with service information
         resource = Resource.create({
             "service.name": settings.SERVICE_NAME,
             "service.version": settings.VERSION,
             "deployment.environment": settings.ENVIRONMENT
         })
         
         # Create tracer provider with the resource
         tracer_provider = TracerProvider(resource=resource)
         
         # Configure exporter
         otlp_endpoint = settings.OTLP_ENDPOINT or "http://jaeger-collector:4318/v1/traces"
         exporter = OTLPSpanExporter(endpoint=otlp_endpoint)
         
         # Add span processor to the tracer provider
         span_processor = BatchSpanProcessor(exporter)
         tracer_provider.add_span_processor(span_processor)
         
         # Set global tracer provider
         trace.set_tracer_provider(tracer_provider)
         
         # Set propagator for distributed tracing context
         propagator = TraceContextTextMapPropagator()
         trace.set_text_map_propagator(propagator)
         
         return tracer_provider
     
     # Helper function to create custom spans
     def create_span(name, context=None):
         tracer = trace.get_tracer(__name__)
         if context:
             return tracer.start_as_current_span(name, context=context)
         return tracer.start_as_current_span(name)
     ```

5. **Custom Middleware for Request Context:**
   * Add middleware to enrich spans with request information:
     ```python
     @app.middleware("http")
     async def add_trace_context_middleware(request: Request, call_next):
         current_span = trace.get_current_span()
         if current_span:
             current_span.set_attribute("http.request.method", request.method)
             current_span.set_attribute("http.request.path", request.url.path)
             # Add user ID if authenticated
             if hasattr(request.state, "user_id"):
                 current_span.set_attribute("user.id", str(request.state.user_id))
         
         response = await call_next(request)
         return response
     ```

6. **Trace Database Operations:**
   * Create a database wrapper in `app/db/tracing.py`:
     ```python
     from opentelemetry import trace
     from sqlalchemy.ext.asyncio import AsyncSession
     
     class TracedAsyncSession(AsyncSession):
         async def execute(self, *args, **kwargs):
             with trace.get_tracer(__name__).start_as_current_span("db.execute") as span:
                 if args and hasattr(args[0], "text"):
                     # Sanitize SQL to remove sensitive data
                     sql_text = str(args[0].text)
                     span.set_attribute("db.statement", sql_text)
                 return await super().execute(*args, **kwargs)
     ```
