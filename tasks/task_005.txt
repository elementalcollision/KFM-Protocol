#### Grace Period Implementation
```python
class GracePeriodManager:
    def __init__(self, config):
        self.config = config
    
    def is_grace_period_expired(self, entity_id, current_state, state_transition_logs):
        """
        Determines if an entity's grace period has expired based on its current state
        and transition history.
        
        Args:
            entity_id: The ID of the entity to check
            current_state: The current EntityState of the entity
            state_transition_logs: Collection of state transition records
            
        Returns:
            bool: True if grace period has expired, False otherwise
        """
        if current_state == EntityState.DEPRECATED:
            # Find when entity was deprecated
            deprecation_time = self._get_latest_transition_time(
                entity_id, EntityState.DEPRECATED, state_transition_logs
            )
            if not deprecation_time:
                return False
                
            grace_period = timedelta(days=self.config.DEFAULT_DEPRECATION_GRACE_PERIOD)
            return datetime.utcnow() > (deprecation_time + grace_period)
            
        elif current_state == EntityState.ARCHIVED:
            # Find when entity was archived
            archival_time = self._get_latest_transition_time(
                entity_id, EntityState.ARCHIVED, state_transition_logs
            )
            if not archival_time:
                return False
                
            retention_period = timedelta(days=self.config.DEFAULT_ARCHIVAL_RETENTION_PERIOD)
            return datetime.utcnow() > (archival_time + retention_period)
            
        return False
    
    def _get_latest_transition_time(self, entity_id, state, transition_logs):
        """
        Retrieves the timestamp of the most recent transition to the specified state.
        
        Implementation will depend on how state_transition_logs are structured.
        """
        relevant_logs = [
            log for log in transition_logs 
            if log.entity_id == entity_id and log.to_state == state
        ]
        
        if not relevant_logs:
            return None
            
        return max(log.transition_time for log in relevant_logs)
```

import httpx
from typing import Dict, Any, Optional
from datetime import datetime

class AgentRegistryClient:
    def __init__(self, base_url: str = "http://agent-registry-service/api/v1"):
        self.base_url = base_url
        
    async def get_agent(self, agent_id: str, token: str) -> Optional[Dict[str, Any]]:
        """Retrieve agent details from the registry"""
        async with httpx.AsyncClient() as client:
            headers = {"Authorization": f"Bearer {token}"}
            response = await client.get(f"{self.base_url}/agents/{agent_id}", headers=headers)
            
            if response.status_code == 404:
                return None
                
            response.raise_for_status()
            return response.json()
            
    async def update_agent_state(self, agent_id: str, new_state: str, token: str, metadata: Dict[str, Any] = None) -> Dict[str, Any]:
        """Update an agent's state in the registry"""
        async with httpx.AsyncClient() as client:
            headers = {"Authorization": f"Bearer {token}"}
            payload = {
                "state": new_state,
                "updated_at": datetime.utcnow().isoformat(),
                "metadata": metadata or {}
            }
            
            response = await client.patch(f"{self.base_url}/agents/{agent_id}/state", json=payload, headers=headers)
            response.raise_for_status()
            return response.json()
            
    async def delete_agent(self, agent_id: str, token: str) -> None:
        """Permanently delete an agent from the registry"""
        async with httpx.AsyncClient() as client:
            headers = {"Authorization": f"Bearer {token}"}
            response = await client.delete(f"{self.base_url}/agents/{agent_id}", headers=headers)
            response.raise_for_status()

def check_permission(required_permission: str) -> Callable:
    """
    Factory function that creates a dependency to check if a user has the required permission.
    
    Args:
        required_permission: The permission string that the user must have
        
    Returns:
        A dependency function that validates the token and permission
        
    Raises:
        HTTPException: If authentication fails or user lacks permission
    """
    async def _check_permission_dependency(token: str = Depends(oauth2_scheme)) -> Tuple[User, str]:

            if not user:
                raise HTTPException(
                    status_code=status.HTTP_401_UNAUTHORIZED,
                    detail="Invalid authentication credentials",
                    headers={"WWW-Authenticate": "Bearer"},
                )
                
            permissions = await get_permissions(user.id)
            if required_permission not in permissions:
                raise HTTPException(
                    status_code=status.HTTP_403_FORBIDDEN,
                    detail=f"Not authorized to perform this operation. Missing permission: {required_permission}",
                )
                
            return user, token
            
        except Exception as e:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail=str(e),
                headers={"WWW-Authenticate": "Bearer"},
            )
            
    return _check_permission_dependency

# k_operator_service/events/publisher.py
import json
import logging
from typing import Any
import aio_pika
from fastapi import Depends
from pydantic import BaseModel

logger = logging.getLogger(__name__)

class EventPublisher:
    def __init__(self):
        self.connection = None
        self.channel = None
        self.exchange = None
        
    async def connect(self, rabbitmq_url: str):
        try:
            self.connection = await aio_pika.connect_robust(rabbitmq_url)
            self.channel = await self.connection.channel()
            self.exchange = await self.channel.declare_exchange(
                "kfm_events", 
                aio_pika.ExchangeType.TOPIC,
                durable=True
            )
            logger.info("Connected to RabbitMQ event bus")
        except Exception as e:
            logger.error(f"Failed to connect to RabbitMQ: {e}")
            raise
            
    async def close(self):
        if self.connection:
            await self.connection.close()
            
    async def publish_event(self, event: BaseModel, routing_key: str):
        if not self.exchange:
            logger.error("Cannot publish event: not connected to RabbitMQ")
            return False
            
        try:
            message_body = json.dumps(event.dict()).encode()
            message = aio_pika.Message(
                body=message_body,
                delivery_mode=aio_pika.DeliveryMode.PERSISTENT,
                content_type="application/json"
            )
            await self.exchange.publish(message, routing_key=routing_key)
            logger.info(f"Published event with routing key: {routing_key}")
            return True
        except Exception as e:
            logger.error(f"Failed to publish event: {e}")
            return False

# FastAPI dependency
async def get_event_publisher(app_state=Depends(get_app_state)):
    return app_state.event_publisher
